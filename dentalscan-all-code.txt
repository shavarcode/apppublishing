      <Stack.Screen name="(tabs)" options={{ headerShown: false }} />
      <Stack.Screen

export default function RootLayout() {
  const [fontsLoaded, fontError] = useFonts({
    Inter_400Regular,
    Inter_500Medium,
    Inter_600SemiBold,
    Inter_700Bold,
  });

  useEffect(() => {
    if (fontsLoaded || fontError) {
      SplashScreen.hideAsync();
    }
  }, [fontsLoaded, fontError]);

  if (!fontsLoaded && !fontError) return null;

  return (
    <SafeAreaProvider>
      <ErrorBoundary>
        <QueryClientProvider client={queryClient}>
          <GestureHandlerRootView style={{ flex: 1 }}>
            <KeyboardProvider>
              <RootLayoutNav />
            </KeyboardProvider>
          </GestureHandlerRootView>
        </QueryClientProvider>
      </ErrorBoundary>
    </SafeAreaProvider>
  );
}

import React, { useState, useCallback } from "react";
import {
  View,
  Text,
  StyleSheet,
  TouchableOpacity,
  ScrollView,
  Platform,
  ActivityIndicator,
  Alert,
  Image,
} from "react-native";
import { useSafeAreaInsets } from "react-native-safe-area-context";
import { useRouter, useFocusEffect } from "expo-router";
import * as ImagePicker from "expo-image-picker";
import * as Haptics from "expo-haptics";
import AsyncStorage from "@react-native-async-storage/async-storage";
import { Ionicons, MaterialCommunityIcons } from "@expo/vector-icons";
import { useColors } from "@/hooks/useColors";

const HISTORY_KEY = "teeth_scan_history";

interface ScanRecord {
  id: string;
  timestamp: number;
  imageUri: string;
  whitenessScore: number;
  shadeGuide: string;
  colorDescription: string;
  colorHex: string;
  overallScore: number;
  urgency: string;
  alignment: string;
  gumHealth: string;
  conditions: Array<{ name: string; severity: string; location: string; advice: string }>;
  recommendations: string[];
  alignmentDetails: string;
  alignmentAdvice: string;
  gumDetails: string;
  urgencyReason: string;
  professionalNote: string;
  confidence: string;
}

const URGENCY_LABELS: Record<string, string> = {
  routine: "Routine",
  attention_needed: "Attention Needed",
  see_dentist_soon: "See Dentist Soon",
  see_dentist_immediately: "See Dentist Immediately",
};

export default function ScannerScreen() {
  const insets = useSafeAreaInsets();
  const colors = useColors();
  const router = useRouter();
  const [analyzing, setAnalyzing] = useState(false);
  const [history, setHistory] = useState<ScanRecord[]>([]);

  useFocusEffect(
    useCallback(() => {
      loadHistory();
    }, [])
  );

  const loadHistory = async () => {
    try {
      const stored = await AsyncStorage.getItem(HISTORY_KEY);
      if (stored) {
        const parsed: ScanRecord[] = JSON.parse(stored);
        setHistory(parsed.sort((a, b) => b.timestamp - a.timestamp));
      }
    } catch {}
  };

  const urgencyColor = (u: string) => {
    if (u === "routine") return colors.urgencyRoutine;
    if (u === "attention_needed") return colors.urgencyAttention;
    if (u === "see_dentist_soon") return colors.urgencySoon;
    return colors.urgencyImmediate;
  };

  const handleScan = async () => {
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);

    let asset: ImagePicker.ImagePickerAsset | null = null;

    if (Platform.OS === "web") {
      const result = await ImagePicker.launchImageLibraryAsync({
        mediaTypes: ["images"],
        quality: 0.85,
        base64: true,
        allowsEditing: true,
        aspect: [4, 3],
      });
      if (result.canceled || !result.assets[0]) return;
      asset = result.assets[0];
    } else {
      const { status } = await ImagePicker.requestCameraPermissionsAsync();
      if (status !== "granted") {
        Alert.alert(
          "Camera Permission Required",
          "Please allow camera access to scan your teeth.",
          [{ text: "OK" }]
        );
        return;
      }
      const result = await ImagePicker.launchCameraAsync({
        mediaTypes: ["images"],
        quality: 0.85,
        base64: true,
        allowsEditing: true,
        aspect: [4, 3],
      });
      if (result.canceled || !result.assets[0]) return;
      asset = result.assets[0];
    }

    if (!asset.base64) {
      Alert.alert("Error", "Could not read image data. Please try again.");
      return;
    }

    setAnalyzing(true);

    try {
      const mimeType = asset.mimeType || "image/jpeg";
      const domain = process.env.EXPO_PUBLIC_DOMAIN;
      const url = domain
        ? `https://${domain}/api/teeth/analyze`
        : "/api/teeth/analyze";

      const response = await fetch(url, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ image: asset.base64, mimeType }),
      });

      if (!response.ok) {
        throw new Error("Analysis failed");
      }

      const analysis = await response.json();

      const record: ScanRecord = {
        id: Date.now().toString() + Math.random().toString(36).substring(2, 7),
        timestamp: Date.now(),
        imageUri: asset.uri,
        ...analysis,
      };

      const stored = await AsyncStorage.getItem(HISTORY_KEY);
      const existing: ScanRecord[] = stored ? JSON.parse(stored) : [];
      const updated = [record, ...existing].slice(0, 50);
      await AsyncStorage.setItem(HISTORY_KEY, JSON.stringify(updated));

      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
      router.push({ pathname: "/result", params: { scanId: record.id } });
    } catch {
      Alert.alert(
        "Analysis Failed",
        "Please ensure your teeth are clearly visible in good lighting and try again."
      );
    } finally {
      setAnalyzing(false);
    }
  };

  const topPad = Platform.OS === "web" ? 67 : insets.top;
  const botPad = Platform.OS === "web" ? 34 : insets.bottom;

  return (
    <View style={[styles.root, { backgroundColor: colors.background }]}>
      <ScrollView
        contentContainerStyle={[
          styles.content,
          { paddingTop: topPad + 20, paddingBottom: botPad + 24 },
        ]}
        showsVerticalScrollIndicator={false}
      >
        {/* Header */}
        <View style={styles.headerRow}>
          <View style={[styles.logoBox, { backgroundColor: colors.primary + "18" }]}>
            <MaterialCommunityIcons name="tooth-outline" size={22} color={colors.primary} />
          </View>
          <View>
            <Text style={[styles.appName, { color: colors.foreground, fontFamily: "Inter_700Bold" }]}>
              DentalScan
            </Text>
            <Text style={[styles.appSub, { color: colors.mutedForeground, fontFamily: "Inter_400Regular" }]}>
              Professional Dental AI
            </Text>
          </View>
        </View>

        {/* Scan card */}
        <View style={[styles.scanCard, { backgroundColor: colors.card, borderColor: colors.border }]}>
          <View style={styles.tipBox}>
            <Ionicons name="information-circle-outline" size={16} color={colors.primary} />
            <Text style={[styles.tipText, { color: colors.mutedForeground, fontFamily: "Inter_400Regular" }]}>
              Smile wide in bright light. Keep the camera steady for the best analysis.
            </Text>
          </View>

          <TouchableOpacity
            style={[
              styles.scanBtn,
              { backgroundColor: analyzing ? colors.muted : colors.primary },
            ]}
            onPress={handleScan}
            disabled={analyzing}
            activeOpacity={0.82}
          >
            {analyzing ? (
              <View style={styles.btnRow}>
                <ActivityIndicator color={colors.mutedForeground} />
                <Text style={[styles.btnText, { color: colors.mutedForeground, fontFamily: "Inter_500Medium" }]}>
                  Analyzing teeth...
                </Text>
              </View>
            ) : (
              <View style={styles.btnRow}>
                <Ionicons
                  name="camera"
                  size={26}
                  color={colors.primaryForeground}
                />
                <Text style={[styles.btnText, { color: colors.primaryForeground, fontFamily: "Inter_600SemiBold" }]}>
                  {Platform.OS === "web" ? "Choose Photo" : "Scan My Teeth"}
                </Text>
              </View>
            )}
          </TouchableOpacity>

          {/* What gets measured */}
          <View style={styles.pillsRow}>
            {["Whiteness", "Shade", "Alignment", "Gum Health", "Conditions"].map((p) => (
              <View key={p} style={[styles.pill, { backgroundColor: colors.secondary }]}>
                <Text style={[styles.pillText, { color: colors.primary, fontFamily: "Inter_500Medium" }]}>
                  {p}
                </Text>
              </View>
            ))}
          </View>
        </View>

        {/* Disclaimer */}
        <View style={[styles.disclaimerBox, { backgroundColor: colors.muted, borderColor: colors.border }]}>
          <Ionicons name="shield-checkmark-outline" size={14} color={colors.mutedForeground} />
          <Text style={[styles.disclaimerText, { color: colors.mutedForeground, fontFamily: "Inter_400Regular" }]}>
            This tool assists with visual dental assessment. It does not replace a clinical diagnosis. Always consult a licensed dentist.
          </Text>
        </View>

        {/* History */}
        {history.length > 0 && (
          <>
            <Text style={[styles.sectionTitle, { color: colors.foreground, fontFamily: "Inter_600SemiBold" }]}>
              Previous Scans
            </Text>
            {history.map((item) => {
              const uc = urgencyColor(item.urgency);
              const sc =
                item.overallScore >= 7
                  ? colors.scoreGood
                  : item.overallScore >= 4
                  ? colors.scoreMid
                  : colors.scoreLow;
              return (
                <TouchableOpacity
                  key={item.id}
                  style={[styles.histRow, { backgroundColor: colors.card, borderColor: colors.border }]}
                  onPress={() => router.push({ pathname: "/result", params: { scanId: item.id } })}
                  activeOpacity={0.75}
                >
                  <Image source={{ uri: item.imageUri }} style={styles.thumb} />
                  <View style={styles.histMid}>
                    <View style={styles.histColorRow}>
                      <View style={[styles.colorDot, { backgroundColor: item.colorHex }]} />
                      <Text style={[styles.histShade, { color: colors.foreground, fontFamily: "Inter_600SemiBold" }]}>
                        {item.shadeGuide} · {item.colorDescription}
                      </Text>
                    </View>
                    <View style={[styles.urgencyTag, { backgroundColor: uc + "18" }]}>
                      <Text style={[styles.urgencyTagText, { color: uc, fontFamily: "Inter_500Medium" }]}>
                        {URGENCY_LABELS[item.urgency] ?? item.urgency}
                      </Text>
                    </View>
                    <Text style={[styles.histDate, { color: colors.mutedForeground, fontFamily: "Inter_400Regular" }]}>
                      {new Date(item.timestamp).toLocaleString("en-US", {
                        month: "short",
                        day: "numeric",
                        hour: "2-digit",
                        minute: "2-digit",
                      })}
                    </Text>
                  </View>
                  <View style={[styles.scoreChip, { backgroundColor: sc + "18" }]}>
                    <Text style={[styles.scoreNum, { color: sc, fontFamily: "Inter_700Bold" }]}>
                      {item.overallScore}
                    </Text>
                    <Text style={[styles.scoreOf, { color: sc, fontFamily: "Inter_400Regular" }]}>/10</Text>
                  </View>
                </TouchableOpacity>
              );
            })}
          </>
        )}

        {history.length === 0 && (
          <View style={[styles.empty, { borderColor: colors.border }]}>
            <MaterialCommunityIcons name="tooth-outline" size={32} color={colors.mutedForeground} />
            <Text style={[styles.emptyText, { color: colors.mutedForeground, fontFamily: "Inter_400Regular" }]}>
              Your dental reports will appear here
            </Text>
          </View>
        )}
      </ScrollView>
    </View>
  );
}

import {

import { useColors } from "@/hooks/useColors";

const HISTORY_KEY = "teeth_scan_history";

interface TeethCondition {
  name: string;
  severity: "none" | "mild" | "moderate" | "severe";
  location: string;
  advice: string;
}

interface ScanRecord {
  id: string;
  timestamp: number;
  imageUri: string;
  whitenessScore: number;
  shadeGuide: string;
  colorDescription: string;

  conditions: TeethCondition[];

}

};

const URGENCY_CONFIG: Record<string, { label: string; icon: string }> = {

  see_dentist_immediately: { label: "See Dentist Immediately", icon: "medical" },

const ALIGNMENT_LABELS: Record<string, string> = {

  mild_crowding: "Mild Crowding",
  moderate_crowding: "Moderate Crowding",
  severe_crowding: "Severe Crowding",
  spacing_issues: "Spacing Issues",
  bite_issues: "Bite Issues",
  other: "Irregular",
};

const GUM_LABELS: Record<string, string> = {

  mild_concern: "Mild Concern",
  moderate_concern: "Moderate Concern",
  severe_concern: "Severe Concern",
};

function SectionHeader({ icon, title, colors }: { icon: string; title: string; colors: any }) {
  return (
    <View style={sectionHeaderStyles.row}>
      <Ionicons name={icon as any} size={16} color={colors.primary} />
      <Text style={[sectionHeaderStyles.text, { color: colors.mutedForeground, fontFamily: "Inter_600SemiBold" }]}>
        {title}
      </Text>

  );

const sectionHeaderStyles = StyleSheet.create({
  row: { flexDirection: "row", alignItems: "center", gap: 6, marginBottom: 2 },

export default function ResultScreen() {
  const insets = useSafeAreaInsets();
  const colors = useColors();
  const router = useRouter();
  const { scanId } = useLocalSearchParams<{ scanId: string }>();
  const [scan, setScan] = useState<ScanRecord | null>(null);

  useEffect(() => { loadScan(); }, [scanId]);
  useEffect(() => {
    if (scan) Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
  }, [scan]);

  const loadScan = async () => {
    try {
      const stored = await AsyncStorage.getItem(HISTORY_KEY);
      if (!stored) return;
      const history: ScanRecord[] = JSON.parse(stored);
      setScan(history.find((s) => s.id === scanId) ?? null);
    } catch {}
  };

  const handleDelete = () => {
    Alert.alert("Delete Report", "Remove this dental report?", [
      { text: "Cancel", style: "cancel" },
      {
        text: "Delete",
        style: "destructive",
        onPress: async () => {
          const stored = await AsyncStorage.getItem(HISTORY_KEY);
          if (!stored) return;
          const history: ScanRecord[] = JSON.parse(stored);
          await AsyncStorage.setItem(HISTORY_KEY, JSON.stringify(history.filter((s) => s.id !== scanId)));
          router.back();
        },
      },
    ]);
  };

  const topPad = Platform.OS === "web" ? 67 : insets.top;
  const botPad = Platform.OS === "web" ? 34 : insets.bottom;

  if (!scan) {
    return (
      <View style={[styles.root, { backgroundColor: colors.background, alignItems: "center", justifyContent: "center" }]}>
        <ActivityIndicatorView colors={colors} />
      </View>
    );

  const urgencyColor =
    scan.urgency === "routine" ? colors.urgencyRoutine :
    scan.urgency === "attention_needed" ? colors.urgencyAttention :
    scan.urgency === "see_dentist_soon" ? colors.urgencySoon :
    colors.urgencyImmediate;

  const scoreColor =
    scan.overallScore >= 7 ? colors.scoreGood :
    scan.overallScore >= 4 ? colors.scoreMid :
    colors.scoreLow;

  const gumColor =
    scan.gumHealth === "healthy" ? colors.urgencyRoutine :
    scan.gumHealth === "mild_concern" ? colors.urgencyAttention :
    scan.gumHealth === "moderate_concern" ? colors.urgencySoon :
    colors.urgencyImmediate;

  const activeConditions = scan.conditions.filter((c) => c.severity !== "none");

  return (
    <View style={[styles.root, { backgroundColor: colors.background }]}>
      <ScrollView
        contentContainerStyle={[
          styles.content,
          { paddingTop: topPad + 8, paddingBottom: botPad + 32 },
        ]}
        showsVerticalScrollIndicator={false}
      >
        {/* Nav bar */}
        <View style={styles.nav}>
          <TouchableOpacity
            style={[styles.navBtn, { backgroundColor: colors.card, borderColor: colors.border }]}
            onPress={() => router.back()}
            activeOpacity={0.75}
          >
            <Ionicons name="chevron-back" size={22} color={colors.foreground} />
          </TouchableOpacity>
          <View>
            <Text style={[styles.navTitle, { color: colors.foreground, fontFamily: "Inter_700Bold" }]}>
              Dental Report
            </Text>
            <Text style={[styles.navDate, { color: colors.mutedForeground, fontFamily: "Inter_400Regular" }]}>
              {new Date(scan.timestamp).toLocaleDateString("en-US", {
                weekday: "short", month: "short", day: "numeric",
                hour: "2-digit", minute: "2-digit",
              })}
            </Text>
          </View>
          <TouchableOpacity
            style={[styles.navBtn, { backgroundColor: "#EF444415" }]}
            onPress={handleDelete}
            activeOpacity={0.75}
          >
            <Ionicons name="trash-outline" size={18} color={colors.destructive} />
          </TouchableOpacity>
        </View>

        {/* Photo */}
        <Image source={{ uri: scan.imageUri }} style={styles.photo} resizeMode="cover" />

        {/* Urgency banner — most important thing at top */}
        <View style={[styles.urgencyBanner, { backgroundColor: urgencyColor + "18", borderColor: urgencyColor + "40" }]}>
          <Ionicons name={urgencyConfig.icon as any} size={22} color={urgencyColor} />
          <View style={{ flex: 1 }}>
            <Text style={[styles.urgencyLabel, { color: urgencyColor, fontFamily: "Inter_700Bold" }]}>
              {urgencyConfig.label}
            </Text>
            <Text style={[styles.urgencyReason, { color: urgencyColor + "CC", fontFamily: "Inter_400Regular" }]}>
              {scan.urgencyReason}
            </Text>
          </View>
        </View>

        {/* 3 top metrics */}
        <View style={styles.metricsRow}>
          {/* Overall score */}
          <View style={[styles.metricCard, { backgroundColor: colors.card, borderColor: colors.border }]}>
            <Text style={[styles.metricBig, { color: scoreColor, fontFamily: "Inter_700Bold" }]}>
              {scan.overallScore}
            </Text>
            <Text style={[styles.metricSub, { color: scoreColor, fontFamily: "Inter_500Medium" }]}>/10</Text>
            <Text style={[styles.metricLabel, { color: colors.mutedForeground, fontFamily: "Inter_500Medium" }]}>OVERALL</Text>

          {/* Whiteness */}
          <View style={[styles.metricCard, { backgroundColor: colors.card, borderColor: colors.border }]}>

Conditions to look for and report (only include if present or suspected):

- "see_dentist_immediately": Severe decay, abscess indicators, severe gum disease, or significant pain signs — see dentist within days`;

router.post("/teeth/analyze", async (req, res) => {
  const { image, mimeType } = req.body as { image: string; mimeType: string };

  if (!image || !mimeType) {
    res.status(400).json({ error: "image and mimeType are required" });
    return;
  }

  try {
    const response = await openai.chat.completions.create({
      model: "gpt-5.4",
      max_completion_tokens: 1024,
      messages: [
        {
          role: "user",
          content: [
            {
              type: "image_url",
              image_url: {
                url: `data:${mimeType};base64,${image}`,
                detail: "high",
              },
            },
            {
              type: "text",
              text: DENTAL_PROMPT,
            },
          ],
        },
      ],
    });

    const content = response.choices[0]?.message?.content;
    if (!content) {
      res.status(500).json({ error: "No response from AI" });
      return;
    }

    const cleaned = content
      .replace(/```json\n?/g, "")
      .replace(/```\n?/g, "")
      .trim();
    const result = JSON.parse(cleaned);

    res.json(result);
  } catch (err) {
    req.log.error({ err }, "Teeth analysis failed");
    res.status(500).json({ error: "Analysis failed. Please try again." });
  }
});

export default router;
